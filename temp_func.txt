function show(id, on) {
  const el = $(id);
  if (!el) return;
  el.style.display = on ? "" : "none";
  // 아코디언 섹션이 처음 표시될 때 자동 펼치기
  if (on && el.classList.contains('collapsible')) {
    el.classList.remove('is-collapsed');
  }
}

function setText(id, val) {
  const el = $(id);
  if (el) el.textContent = val != null ? String(val) : "-";
}
